# Use clean LTS Node.js slim alpine as base
FROM node:20-slim AS builder

WORKDIR /app

# Copy dependency configs
COPY package*.json ./

# Install all dependencies (including devDependencies for esbuild/typescript compilation)
RUN npm ci

# Copy the rest of the application files
COPY . .

# Compile frontend static files and bundle backend to dist/server.cjs
RUN npm run build

# Use lightweight Node runtime for production stage
FROM node:20-slim AS runner

WORKDIR /app

ENV NODE_ENV=production
ENV PORT=3000

# Copy necessary production artifacts from builder stage
COPY --from=builder /app/package*.json ./
COPY --from=builder /app/dist ./dist

# Install ONLY production dependencies to keep the container small and fast
RUN npm ci --omit=dev

# Inform container orchestration platforms that our service binds to port 3000
EXPOSE 3000

# Run the bundled server launcher
CMD ["npm", "run", "start"]
